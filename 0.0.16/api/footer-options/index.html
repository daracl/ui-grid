FooterOptions

Configures the footer bar: paging display and selection summary.

Source: src/types/GridOptions.d.ts, src/defaultGridOption.ts

Type

interface FooterOptions {
  enabled?: boolean;
  height?: number;
  paging?: {
    enabled?: boolean;
    position?: HorizontalRegion;
    formatPosition?: HorizontalRegion;
    format?: string | OptionCallback;
    callback?: OptionCallback;
  };
  selection?: {
    position?: HorizontalRegion;
    format: string | OptionCallback;
    showSummary?: boolean;
  };
}

Properties

Example

footer: {
  enabled: true,
  height: 36,
  paging: { enabled: true, position: 'center', format: '{{start}} - {{end}} of {{total}}' },
  selection: { position: 'left', format: 'Count : {{count}}', showSummary: true },
}
