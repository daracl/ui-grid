API Reference

GridOptions

The full option object passed to `new DaraGrid(element, options)` / `DaraGrid.create(element, options)`. Every other option interface below is a sub-property of GridOptions.

Source: src/types/GridOptions.d.ts, src/defaultGridOption.ts

Type

interface GridOptions {
  rowIdField?: string;
  theme: THEME_TYPE;                 // 'light' | 'dark'
  height: 'auto' | number;
  width: 'auto' | number;
  windowResizeDelay: number;
  fixedLeftIndex: number;
  fixedRightIndex: number;
  style: BodyStyle;                  // 'default' | 'striped' | 'borderless' | 'rows' | 'cells' | 'stripedRows' | 'stripedCells'
  enabledWidthFixed: boolean;
  useDefaultFormatter: boolean;
  selectionMode: SelectionMode;      // 'row' | 'cell' | 'multiRow' | 'multiCell' | 'none'
  hoverMode?: HoverMode;             // 'row' | 'cell' | 'none'
  deleteMode?: DeleteMode;           // 'soft' | 'hard'
  enabledTooltip: boolean;
  addLimitRow: number;
  valueFilter: boolean | OptionCallback;
  dataTypeFormatter: { money: ValueFormatter; number: ValueFormatter };
  editable: boolean;
  header: HeaderOptions;
  search: SearchOptions;
  toolbar: ToolbarOptions;
  aside: AsideOptions;
  body: BodyOptions;
  scroll: ScrollOptions;
  paging?: PagingParam;
  footer: FooterOptions;
  i18n: any;
  operators: any;
  fields: FieldItem[];
  items: any[];
  summary?: SummaryOptions;
  contextMenu?: ContextMenuOptions;
  tree?: TreeOptions;
}

Example

import { DaraGrid } from '@daracl/grid';
import type { GridOptions } from '@daracl/grid';
import '@daracl/grid/style/daracl.grid.css';

const grid = DaraGrid.create(document.getElementById('grid')!, {
  height: 400,
  editable: true,
  selectionMode: 'multiCell',
  theme: 'light',
  style: 'striped',
  fields: [
    { name: 'name', label: 'Name', width: 100 },
    { name: 'age', label: 'Age', width: 60, align: 'right', renderer: { type: 'number' } },
  ],
  items: [
    { name: 'John', age: 20 },
    { name: 'Jane', age: 25 },
  ],
} as GridOptions);
