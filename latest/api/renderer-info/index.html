API Reference

RendererInfo (View / Edit)

Base shape for field.renderer (view) and field.editRenderer (edit), plus the concrete `type` values registered in constantRenders.ts.

Source: src/types/RendererInfo.d.ts, src/constantRenders.ts, src/constantStyles.ts

Type

interface RendererInfo {
  type: string;
  variant?: string;
  click?: OptionCallback;
}

interface ViewRendererInfo extends RendererInfo {
  tooltip?: string;
  refValue?: OptionCallback | any;
  rule?: { minLength: number; maxLength: number; minimum: number; maximum: number };
}

interface EditRendererInfo extends RendererInfo {
  customOptions?: any;
  required?: boolean;
  regexpType?: RegexpType;          // 'email' | 'url' | 'alpha' | 'alpha-num'
  rule?: {
    minLength: number; maxLength: number;
    minimum: number; exclusiveMinimum: boolean;
    maximum: number; exclusiveMaximum: boolean;
  };
  different?: { field: string; message: string };
  identical?: { field: string; message: string };
  defaultValue?: string;
  listItem?: ValuesInfo;
  validator?: OptionCallback;
  change?: OptionCallback;
  trueValue?: string | boolean;
  falseValue?: string | boolean;
  showLabel?: boolean;
  iconStyle?: string;
  icon?: string;
}
