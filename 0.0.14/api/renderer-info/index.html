RendererInfo (View / Edit)

Base shape for field.renderer (view) and field.editRenderer (edit), plus the concrete `type` values registered in constantRenders.ts.

Source: src/types/RendererInfo.d.ts, src/constantRenders.ts, src/constantStyles.ts

Type

interface RendererInfo {
  type: string;
  variant?: string;
  click?: OptionCallback;
  conditional?: ConditionInfo;
}

interface ViewRendererInfo extends RendererInfo {
  tooltip?: string;
  refValue?: OptionCallback | any;
  rule?: { minLength: number; maxLength: number; minimum: number; maximum: number };
}

interface EditRendererInfo extends RendererInfo {
  customOptions?: any;
  required?: boolean;
  regexpType?: RegexpType;          // 'email' | 'url' | 'alpha' | 'alpha-num'
  rule?: {
    minLength: number; maxLength: number;
    minimum: number; exclusiveMinimum: boolean;
    maximum: number; exclusiveMaximum: boolean;
  };
  different?: { field: string; message: string };
  identical?: { field: string; message: string };
  defaultValue?: string;
  listItem?: ValuesInfo;
  validator?: OptionCallback;
  change?: OptionCallback;
  trueValue?: string | boolean;
  falseValue?: string | boolean;
  showLabel?: boolean;
  iconStyle?: string;
  icon?: string;
}

Properties

PropertyTypeDefaultDescription
typestring—Renderer type key. View: 'text' | 'number' | 'password' | 'dropdown' | 'bar' | 'button' | 'html' | 'image' | 'link' | 'sparkline' | 'sparklineBar' | 'tree' | 'custom'. Edit: 'text' | 'number' | 'password' | 'textarea' | 'dropdown' | 'checkbox' | 'switch' | 'range' | 'date' | 'custom'.
variant?string—Predefined style variant for renderers that support it. Currently only 'switch' does, with 'box' | 'round' (see constantStyles.ts RENDERER_VARIANTS).
click?OptionCallback—Click handler for clickable renderers (e.g. button, link).
listItem?ValuesInfo—Option list for dropdown/choice-style renderers: { labelField, valueField, list, multiple, orientation, includeAllOption }.
trueValue / falseValue?string | booleantrue / falseValues written back for checkbox/switch renderers.
showLabel?booleanfalseShows the on/off label text inside switch-style renderers.
validator?OptionCallback—Custom validation function for edit renderers.
required?boolean—Marks the edit renderer as required for validation.
regexpType?'email' | 'url' | 'alpha' | 'alpha-num'—Built-in regex validation preset.
