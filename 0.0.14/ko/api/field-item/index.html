API 레퍼런스

GridField (FieldItem)

하나의 컬럼을 정의합니다: 라벨, 너비, 정렬, 보기/편집 렌더러 및 동작. GridOptions.fields로 전달합니다.

소스: src/types/GridField.d.ts, src/defaultGridOption.ts (DEFAULT_FIELD_INFO)

타입

interface FieldItem {
  name: string;
  label: string;
  getValue?: OptionCallback;
  width: number;
  colspan: number;
  rowspan: number;
  hidden: boolean;
  editable: boolean;
  sort: boolean;
  align: TextAlignType;              // 'left' | 'center' | 'right'
  whiteSpace?: WhiteSpaceType;        // 'pre' | 'wrap' | 'line' | 'normal'
  renderer: ViewRendererInfo;
  editRenderer: EditRendererInfo;
  displayFormat?: DisplayFormatOptions;
  defaultValue: string;
  cellClassName?: OptionCallback | string;
  tooltip?: { enabled: boolean; content?: OptionCallback };
  headerHelp: string | OptionCallback | undefined;
  children?: FieldItem[];
  // Internal, computed at runtime — do not set these yourself:
  // $renderer, $editRenderer, $colspan, $rowspan, $depth, $isLeaf,
  // $childLength, $resizeIdx, $width, $alignStyle, $isAside, $panel, $uid, $colSeq, $enabledHelp
}
